#pragma once

#include <string>
#include <vector>
#include <unordered_map>
#include <functional>
#include <iostream>

// --------------------
// Speaker + screen pos
// --------------------
enum class Speaker {
    Narrator,
    Witch1,
    Witch2,
    Player
};

enum class ScreenPos {
    Left,
    Center,
    Right
};

// --------------------
// Dialogue Line
// --------------------
struct DialogueLine {
    Speaker speaker;
    std::string text;
    std::string sprite;      // sprite ID
    ScreenPos position;
};

// --------------------
// Choice
// --------------------
struct Choice {
    std::string text;
    std::string nextNode;
    std::function<bool()> condition = nullptr;

    bool isAvailable() const {
        return !condition || condition();
    }
};

// --------------------
// Node (VN dialogue block)
// --------------------
struct Node {
    std::string id;
    std::vector<DialogueLine> lines;
    std::vector<Choice> choices;
    std::function<void()> onEnter = nullptr;
};

// --------------------
// Plot state (runtime)
// --------------------
struct PlotState {
    std::string currentNode;
    size_t currentLine = 0;
};

// --------------------
// Plot graph
// --------------------
class PlotGraph {
public:
    std::unordered_map<std::string, Node> nodes;

    // Advance one line, return false if node finished
    bool advance(PlotState& state) {
        if (nodes.find(state.currentNode) == nodes.end()) {
            std::cerr << "Node not found: " << state.currentNode << "\n";
            return false;
        }

        Node& node = nodes[state.currentNode];

        if (state.currentLine == 0 && node.onEnter)
            node.onEnter();

        if (state.currentLine < node.lines.size()) {
            const DialogueLine& line = node.lines[state.currentLine];
            // For now, just print (replace with UI hook)
            std::cout << "[" << static_cast<int>(line.speaker) << "] ";
            std::cout << line.text << "\n";
            state.currentLine++;
            return true;
        }

        return false;
    }

    // Get available choices for current node
    std::vector<Choice> getAvailableChoices(PlotState& state) {
        Node& node = nodes[state.currentNode];
        std::vector<Choice> available;

        for (const auto& c : node.choices)
            if (c.isAvailable())
                available.push_back(c);

        return available;
    }

    // Select a choice, moves to next node
    void selectChoice(PlotState& state, size_t choiceIndex) {
        Node& node = nodes[state.currentNode];
        std::vector<Choice> available = getAvailableChoices(state);

        if (choiceIndex >= available.size()) return;

        state.currentNode = available[choiceIndex].nextNode;
        state.currentLine = 0;
    }
};
